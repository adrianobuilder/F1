    class OpenF1Client:
    """Reutiliza conexoes e limita tentativas de falhas transitorias."""

    def __init__(
        self,
        settings: Settings,
        transport: httpx.BaseTransport | None = None,
        sleep: Callable[[float], None] = time.sleep,
    ) -> None:
        self.settings = settings
        self.sleep = sleep
        self.http = httpx.Client(
            base_url="https://api.openf1.org/v1/",
            timeout=settings.timeout,
            transport=transport,
            headers={"User-Agent": "f1-historical-dashboard/0.1"},
        )

    def close(self) -> None:
        """Libera conexoes do cliente."""
        self.http.close()

    def fetch(self, endpoint: Endpoint, params: dict[str, int]) -> list[Record]:
        """Retorna objetos JSON ou falha explicitamente, inclusive em HTTP 4xx.

        Args:
            endpoint: Recurso permitido pelo dominio.
            params: Filtros de temporada ou sessao.

        Raises:
            SourceError: Em falha de rede, HTTP ou estrutura JSON invalida.
        """
        for attempt in range(self.settings.attempts):
            try:
                response = self.http.get(endpoint.value, params=params)
            except httpx.TransportError as exc:
                if attempt + 1 == self.settings.attempts:
                    raise SourceError(
                        "A OpenF1 nao respondeu. Tente novamente."
                    ) from exc
                self.sleep(2**attempt + random.uniform(0, 0.5))
                continue
            if (
                response.status_code in {429, 500, 502, 503, 504}
                and attempt + 1 < self.settings.attempts
            ):
                self.sleep(self._delay(response, attempt))
                continue
            try:
                response.raise_for_status()
                payload = response.json()
            except (httpx.HTTPStatusError, ValueError) as exc:
                raise SourceError(
                    f"Consulta OpenF1 falhou ({response.status_code})."
                ) from exc
            if not isinstance(payload, list) or any(
                not isinstance(row, dict) for row in payload
            ):
                raise SourceError("A OpenF1 retornou um formato inesperado.")
            return cast(list[Record], payload)
        raise SourceError("Consulta OpenF1 indisponivel.")

    @staticmethod
    def _delay(response: httpx.Response, attempt: int) -> float:
        value = response.headers.get("Retry-After", "")
        try:
            delay = float(value)
        except ValueError:
            try:
                delay = (
                    parsedate_to_datetime(value) - datetime.now(UTC)
                ).total_seconds()
            except (TypeError, ValueError, OverflowError):
                delay = 2**attempt + random.uniform(0, 0.5)
        if delay > 60:
            raise SourceError("Limite da OpenF1 atingido. Aguarde antes de atualizar.")
        return max(0, delay)

    class SnapshotStore:
    """Persiste geracoes sem sobrescrever o ultimo snapshot valido."""

    def __init__(self, root: Path) -> None:
        self.root = root.resolve()

    def _manifest(self, endpoint: Endpoint, params: dict[str, int]) -> Path:
        key = "_".join(f"{key}-{value}" for key, value in sorted(params.items()))
        if any(key not in {"year", "session_key"} for key in params) or any(
            not isinstance(value, int) or value < 1 for value in params.values()
        ):
            raise StorageError("Particao de armazenamento invalida.")
        return self.root / "manifests" / endpoint.value / f"{key}.json"

    def read(
        self, endpoint: Endpoint, params: dict[str, int]
    ) -> tuple[pd.DataFrame, Provenance] | None:
        """Le somente geracoes completas da versao de schema atual."""
        manifest_path = self._manifest(endpoint, params)
        if not manifest_path.exists():
            return None
        try:
            manifest = json.loads(manifest_path.read_text(encoding="utf-8"))
            if manifest["schema_version"] != SCHEMA_VERSION:
                return None
            if manifest["state"] != "complete" or manifest["params"] != params:
                raise ValueError("Manifesto inconsistente")
            target = (self.root / manifest["processed"]).resolve()
            if not target.is_relative_to(self.root):
                raise ValueError("Caminho invalido")
            if self._digest(target) != manifest["sha256"]:
                raise ValueError("Checksum invalido")
            frame = pd.read_parquet(target)
            if len(frame) != manifest["rows"]:
                raise ValueError("Contagem inconsistente")
            return frame, Provenance(datetime.fromisoformat(manifest["collected_at"]))
        except (OSError, ValueError, KeyError, TypeError) as exc:
            raise StorageError("Snapshot local invalido; atualize os dados.") from exc

    def publish(
        self,
        endpoint: Endpoint,
        params: dict[str, int],
        raw: list[Record],
        frame: pd.DataFrame,
    ) -> Provenance:
        """Publica manifesto somente apos JSON e Parquet completos.

        Uma falha deixa no maximo arquivos orfaos; a geracao anterior permanece
        referenciada. Nao ha garantia de coordenacao entre escritores concorrentes.
        """
        manifest_path = self._manifest(endpoint, params)
        generation = uuid4().hex
        scope = manifest_path.stem
        relative = Path(endpoint.value) / scope / generation
        raw_path = self.root / "raw" / relative.with_suffix(".json")
        processed_path = self.root / "processed" / relative.with_suffix(".parquet")
        collected = datetime.now(UTC)
        temporary = manifest_path.with_suffix(f".{generation}.tmp")
        try:
            for path in (raw_path, processed_path, manifest_path):
                path.parent.mkdir(parents=True, exist_ok=True)
            raw_path.write_text(
                json.dumps(raw, ensure_ascii=False, allow_nan=False), encoding="utf-8"
            )
            frame.to_parquet(processed_path, index=False)
            manifest = {
                "endpoint": endpoint.value,
                "params": params,
                "schema_version": SCHEMA_VERSION,
                "state": "complete",
                "collected_at": collected.isoformat(),
                "rows": len(frame),
                "raw": raw_path.relative_to(self.root).as_posix(),
                "processed": processed_path.relative_to(self.root).as_posix(),
                "sha256": self._digest(processed_path),
            }
            temporary.write_text(json.dumps(manifest), encoding="utf-8")
            temporary.replace(manifest_path)
        except (OSError, ValueError, TypeError) as exc:
            raise StorageError("Nao foi possivel publicar os arquivos locais.") from exc
        finally:
            temporary.unlink(missing_ok=True)
        return Provenance(collected)

    @staticmethod
    def _digest(path: Path) -> str:
        with path.open("rb") as stream:
            return cast(str, hashlib.file_digest(stream, "sha256").hexdigest())

    def normalize(endpoint: Endpoint, raw: list[Record]) -> pd.DataFrame:
    """Valida chaves, normaliza datas/tipos e rejeita duplicatas conflitantes.

    Campos nao previstos permanecem no bruto. Arrays nao analiticos sao texto
    JSON no Parquet; duracao e diferencas de classificacao possuem colunas Q1-Q3.
    """
    frame = pd.DataFrame([_flatten(row) for row in raw])
    keys = KEYS[endpoint]
    if frame.empty:
        return pd.DataFrame({key: pd.Series(dtype="object") for key in keys})
    if any(key not in frame or frame[key].isna().any() for key in keys):
        raise ContractError(f"Campos essenciais invalidos em {endpoint.value}.")
    required = {
        Endpoint.MEETINGS: {"meeting_name"},
        Endpoint.SESSIONS: {"meeting_key", "session_name", "date_start", "date_end"},
        Endpoint.POSITION: {"position"},
    }.get(endpoint, set())
    if not required.issubset(frame.columns):
        raise ContractError(f"Schema incompleto em {endpoint.value}.")
    _normalize_types(frame, endpoint)
    _validate_keys(frame, keys)
    if (
        "is_pit_out_lap" in frame
        and not frame["is_pit_out_lap"]
        .dropna()
        .map(lambda value: isinstance(value, bool))
        .all()
    ):
        raise ContractError("Indicador de volta de saida invalido.")
    frame = frame.drop_duplicates()
    if frame.duplicated(keys).any():
        raise ContractError(f"Registros conflitantes em {endpoint.value}.")
    return frame.sort_values(keys).reset_index(drop=True)

    class DataService:
    """Coordena cliente e snapshots; nao depende do Streamlit."""

    def __init__(
        self, client: OpenF1Client, store: SnapshotStore, settings: Settings
    ) -> None:
        self.client, self.store, self.settings = client, store, settings

    def dataset(
        self, endpoint: Endpoint, params: dict[str, int], *, refresh: bool = False
    ) -> Dataset:
        """Reutiliza dados completos; atualiza catalogos e preserva fallback."""
        previous = None
        try:
            previous = self.store.read(endpoint, params)
        except StorageError:
            logger.warning("snapshot_unreadable", extra={"endpoint": endpoint.value})
        catalog = endpoint in {Endpoint.MEETINGS, Endpoint.SESSIONS}
        if previous and not refresh:
            frame, provenance = previous
            expired = catalog and datetime.now(
                UTC
            ) - provenance.collected_at > timedelta(
                hours=self.settings.catalog_ttl_hours
            )
            if not expired:
                return Dataset(frame, provenance)
        run_id, started = uuid4().hex, monotonic()
        context = {"run_id": run_id, "endpoint": endpoint.value}
        logger.info("dataset_started", extra=context)
        try:
            raw = self.client.fetch(endpoint, params)
            frame = normalize(endpoint, raw)
            if (
                "session_key" in params
                and "session_key" in frame
                and not frame["session_key"].eq(params["session_key"]).all()
            ):
                raise F1Error("A resposta inclui outra sessao.")
            provenance = self.store.publish(endpoint, params, raw, frame)
        except F1Error:
            logger.exception("dataset_failed", extra=context)
            if previous is None:
                raise
            frame, provenance = previous
            return Dataset(
                frame,
                replace(
                    provenance,
                    stale=True,
                    warning="Atualizacao falhou. Exibindo o ultimo snapshot valido.",
                ),
            )
        logger.info(
            "dataset_completed",
            extra={
                **context,
                "rows": len(frame),
                "duration_seconds": monotonic() - started,
            },
        )
        return Dataset(frame, provenance)

def open_service(settings: Settings) -> Iterator[DataService]:

    try:

    def completed_sessions(
    sessions: pd.DataFrame, now: datetime | None = None
) -> pd.DataFrame:
    """Seleciona sessoes encerradas e nao canceladas, mais recentes primeiro."""
    if sessions.empty or "date_end" not in sessions:
        return sessions.iloc[:0].copy()
    end = pd.to_datetime(sessions["date_end"], utc=True, errors="coerce")
    cancelled = sessions.get("is_cancelled", pd.Series(False, index=sessions.index))
    mask = (end <= (now or datetime.now(UTC))) & ~cancelled.fillna(False).astype(bool)
    return sessions.loc[mask].sort_values("date_end", ascending=False).copy()

    def event_time(value: object, offset: str) -> str:
    """Formata um instante no deslocamento UTC fornecido pelo evento."""
    try:
        sign = -1 if offset.startswith("-") else 1
        hours, minutes, *_ = offset.lstrip("+-").split(":")
        tz = timezone(sign * timedelta(hours=int(hours), minutes=int(minutes)))
        timestamp = pd.Timestamp(str(value))
        if timestamp.tzinfo is None:
            timestamp = timestamp.tz_localize("UTC")
        return timestamp.tz_convert(tz).strftime("%d/%m/%Y %H:%M (UTC%z)")
    except (ValueError, TypeError):
        return "Horario indisponivel"

    def format_lap(value: float) -> str:
    """Formata segundos como minutos, segundos e milissegundos."""
    milliseconds = round(value * 1000)
    minutes, remainder = divmod(milliseconds, 60000)
    seconds, millis = divmod(remainder, 1000)
    return f"{minutes}:{seconds:02d}.{millis:03d}"

    def lap_series(laps: pd.DataFrame, drivers: list[int]) -> pd.DataFrame:
    """Preserva lacunas de voltas ausentes ou excluidas nos graficos."""
    frame = filter_drivers(laps, drivers)
    if frame.empty or "lap_duration" not in frame:
        return frame.iloc[:0]
    pit_out = frame.get("is_pit_out_lap", pd.Series(False, index=frame.index))
    valid = frame["lap_duration"].gt(0) & ~pit_out.astype("boolean").fillna(False)
    durations = [
        column for column in frame if str(column).startswith("duration_sector_")
    ]
    for column in ["lap_duration", *durations]:
        frame[column] = frame[column].where(valid)
    groups = []
    for number, group in frame.groupby("driver_number"):
        first, last = int(group["lap_number"].min()), int(group["lap_number"].max())
        indexed = group.set_index("lap_number").reindex(range(first, last + 1))
        indexed["driver_number"] = number
        indexed.index.name = "lap_number"
        groups.append(indexed.reset_index())
    return pd.concat(groups, ignore_index=True)

    def localize_times(frame: pd.DataFrame, offset: str) -> pd.DataFrame:
    """Converte somente a copia de apresentacao; armazenamento permanece UTC."""
    result = frame.copy()

            result[column] = pd.to_datetime(result[column], utc=True).dt.tz_convert(tz)
    return result

class Settings:
    """Parametros operacionais; caminhos sao relativos ao diretorio de execucao."""

    data_dir: Path = Path("data")
    log_level: str = "INFO"
    timeout: float = 20
    attempts: int = 3
    catalog_ttl_hours: float = 24

    @classmethod
    def from_env(cls, env: Mapping[str, str] | None = None) -> "Settings":
        """Le e valida os limites antes de abrir conexoes ou criar arquivos."""
        values = os.environ if env is None else env
        try:
            settings = cls(
                data_dir=Path(values.get("F1_DATA_DIR", "data")),
                log_level=values.get("F1_LOG_LEVEL", "INFO").upper(),
                timeout=float(values.get("F1_HTTP_TIMEOUT", "20")),
                attempts=int(values.get("F1_HTTP_ATTEMPTS", "3")),
                catalog_ttl_hours=float(values.get("F1_CATALOG_TTL_HOURS", "24")),
            )
        except (ValueError, TypeError) as exc:
            raise ValueError("Configuracao F1 numerica invalida") from exc
        if not 0 < settings.timeout <= 120 or not 1 <= settings.attempts <= 3:
            raise ValueError("Timeout deve ser (0,120] e tentativas entre 1 e 3")
        if settings.catalog_ttl_hours <= 0 or settings.log_level not in (
            "DEBUG",
            "INFO",
            "WARNING",
            "ERROR",
            "CRITICAL",
        ):
            raise ValueError("TTL ou nivel de log invalido")
        return settings

    def logger_level(self) -> int:
        """Converte o nivel validado para a biblioteca padrao."""
        return int(logging.getLevelNamesMapping()[self.log_level])
